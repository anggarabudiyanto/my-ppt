          \end{lstlisting}
  \end{itemize}

%         matriks (bmatrix), fungsi tr (DeclareMathOperator)
%-------------------------------------------------------------------------------
\begin{frame}
  \frametitle{Menulis Persamaan}

  % (1) MATEMATIKA DALAM KALIMAT : diapit \sel{...} atau $...$
 %      contoh: entropi maksimum dua keadaan adalah \sel{S = 1} bit.
  Entropi maksimum satu qubit adalah \sel{S\!\left(\tfrac{1}{2}I\right) = 1} bit,
  sedangkan keadaan murni punya \sel{S(\ket{\psi}) = 0}.

  \medskip
  % (2) PERSAMAAN BERNOMOR : equation + \label{eq:nama}; dirujuk \equ~\ref{eq:...}
  %     \equ adalah makro yang mencetak kata "Persamaan" (ppt.sty).
  \begin{equation}
    S(\rho) = -\tr\!\left(\rho \log_2 \rho\right)
    \label{eq:entropi-vonneumann}
  \end{equation}
  \equ~\ref{eq:entropi-vonneumann} adalah entropi von Neumann untuk matriks
  densitas \sel{\rho}; huruf \texttt{\textbackslash tr} dicetak tegak karena
  didefinisikan dengan \texttt{\textbackslash DeclareMathOperator}.

  \medskip
  % (3) PERSAMAAN BANYAK BARIS : align* (tanpa nomor) — & menandai titik
  %     penyelarasan, \\ memindah baris. Gunakan align (tanpa bintang) bila
  %     setiap baris ingin diberi nomor.
  \begin{align*}
    \ket{\psi} &= \cos\!\left(\frac{\theta}{2}\right)\ket{0}
                 + e^{i\phi}\sin\!\left(\frac{\theta}{2}\right)\ket{1} \\
    \rho &= \ket{\psi}\bra{\psi}
          = \frac{1}{2}
            \begin{bmatrix}
              1 & e^{-i\phi} \\
              e^{i\phi} & 1
            \end{bmatrix}
  \end{align*}

  % (4) Kasus terpisah (kurung kurawal) bila dibutuhkan:
  %     \begin{cases} ... & ... \\ ... & ... \end{cases}  (di dalam equation)
\end{frame}

\section{Notasi Kuantum}

%-------------------------------------------------------------------------------
% SLIDE : "Notasi Dirac (Ket-Bra)"
% Fitur : \ket \bra \braket \ketsub (dari paket braket + makro ppt.sty)
%-------------------------------------------------------------------------------
\begin{frame}[fragile]
  \frametitle{Notasi Dirac (Ket--Bra)}

  % PAKET braket menyediakan:  \ket{x} -> |x⟩   \bra{x} -> ⟨x|
  %   \braket{a|b} -> ⟨a|b⟩    \ket{a}\!\bra{b} -> |a⟩⟨b|
  % MAKRO \ketsub{psi}{AB} (ppt.sty) menuliskan |ψ⟩_AB — subskrip sistem.
  % PENTING: subskrip diletakkan DI LUAR \ket{...}, bukan di dalam kurungnya:
  %   BENAR : \ket{\psi}_{AB}   SALAH : \ket{\psi_{AB}}  (subskrip ikut tercetak
  %             di dalam ket, tipografi & maknanya keliru)
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Keadaan dasar qubit:
              \sel{\ket{0},\ \ket{1}}
        \item Superposisi (Bell):
              \sel{\ketsub{\psi}{AB} = \tfrac{1}{\sqrt{2}}\left(\ket{00}+\ket{11}\right)}
        \item Proyeksi \& dual:
              \sel{\bra{0},\ \ket{0}\!\bra{0},\ \braket{0|1} = 0}
        \item Operator evolusi unitari:
              \sel{\hat{U}\hat{U}^{\dagger} = \hat{I} = \hat{U}^{\dagger}\hat{U}}
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      % Contoh penerapan lengkap: matriks densitas & partial trace.
      % {\small ...} mengecilkan font HANYA untuk blok persamaan ini agar muat.
      {\small
      Matriks densitas dan \istilah{partial trace}:
      \begin{align*}
        \rho_{AB} &= \ketsub{\psi}{AB}\brasub{\psi}{AB} \\
        \rho_A    &= \tr_B(\rho_{AB}) \equiv \tfrac{1}{2} I_A \\
        S(\rho_A) = S(\rho_B) &= 1 \\
        S(\rho_A|\rho_B) &= S(\rho_{AB}) - S(\rho_B) = -1
      \end{align*}}
      \begin{lstlisting}[basicstyle=\ttfamily\tiny]
\ketsub{\psi}{AB}      % |psi>_AB
\brasub{\psi}{AB}      % <psi|_AB
\rho = \ket{\psi}\bra{\psi}
\tr_B(\rho_{AB})       % partial trace
\hat{U}^{\dagger}      % transpose konjugat
      \end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

ket0 = np.array([1, 0])   # |0>

H = np.array([[1,  1],    # Hadamard
              [1, -1]]) / np.sqrt(2)

prob = np.abs(psi)**2     # peluang ukur

    \item \textbf{\texttt{\textbackslash secondcite}} --- mengutip ulang, tampil nomornya saja:
          kodetan Slepian--Wolf\secondcite{slepian1973noiseless}.
