\documentclass[
    11pt,           % ukuran font dasar (boleh 10pt / 12pt)
    aspectratio=169 % rasio layar 16:9 (hapus baris ini untuk rasio 4:3)

\begin{document}

\input{01-sampul}         % 1. halaman judul

\input{04-referensi}      % daftar pustaka (otomatis dari bib.bib)
\input{99-penutup}        % slide penutup

\end{document}
